cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(rf_actuated_boot_entropy)

set(WIFI_SSID_VALUE $ENV{WIFI_SSID})
set(WIFI_PASS_VALUE $ENV{WIFI_PASS})
if(DEFINED WIFI_SSID)
    set(WIFI_SSID_VALUE ${WIFI_SSID})
endif()
if(DEFINED WIFI_PASS)
    set(WIFI_PASS_VALUE ${WIFI_PASS})
endif()
if(NOT WIFI_SSID_VALUE)
    set(WIFI_SSID_VALUE "WIFI_SSID_NOT_SET")
endif()
if(NOT WIFI_PASS_VALUE)
    set(WIFI_PASS_VALUE "WIFI_PASS_NOT_SET")
endif()

set(AEB_UDP_PORT_VALUE $ENV{AEB_UDP_PORT})
if(DEFINED AEB_UDP_PORT)
    set(AEB_UDP_PORT_VALUE ${AEB_UDP_PORT})
endif()
if(NOT AEB_UDP_PORT_VALUE)
    set(AEB_UDP_PORT_VALUE "7777")
endif()

set(AEB_GATEWAY_IP_VALUE $ENV{AEB_GATEWAY_IP})
if(DEFINED AEB_GATEWAY_IP)
    set(AEB_GATEWAY_IP_VALUE ${AEB_GATEWAY_IP})
endif()
if(NOT AEB_GATEWAY_IP_VALUE)
    set(AEB_GATEWAY_IP_VALUE "0.0.0.0")
endif()

set(AEB_GATEWAY_PORT_VALUE $ENV{AEB_GATEWAY_PORT})
if(DEFINED AEB_GATEWAY_PORT)
    set(AEB_GATEWAY_PORT_VALUE ${AEB_GATEWAY_PORT})
endif()
if(NOT AEB_GATEWAY_PORT_VALUE)
    set(AEB_GATEWAY_PORT_VALUE "7778")
endif()

    set(AEB_CLIENT_INITIATED_VALUE ${AEB_CLIENT_INITIATED})
endif()
if(NOT AEB_CLIENT_INITIATED_VALUE)
    set(AEB_CLIENT_INITIATED_VALUE "1")
endif()

set(AEB_CLIENT_TRIALS_VALUE $ENV{AEB_CLIENT_TRIALS})
if(DEFINED AEB_CLIENT_TRIALS)
    set(AEB_CLIENT_TRIALS_VALUE ${AEB_CLIENT_TRIALS})
endif()
if(NOT AEB_CLIENT_TRIALS_VALUE)
    set(AEB_CLIENT_TRIALS_VALUE "16")
endif()

set(AEB_BURST_COUNT_VALUE $ENV{AEB_BURST_COUNT})
if(DEFINED AEB_BURST_COUNT)
    set(AEB_BURST_COUNT_VALUE ${AEB_BURST_COUNT})
endif()
if(NOT AEB_BURST_COUNT_VALUE)
    set(AEB_BURST_COUNT_VALUE "64")
endif()

set(AEB_INTERVAL_US_VALUE $ENV{AEB_INTERVAL_US})
if(DEFINED AEB_INTERVAL_US)
    set(AEB_INTERVAL_US_VALUE ${AEB_INTERVAL_US})
endif()
if(NOT AEB_INTERVAL_US_VALUE)
    set(AEB_INTERVAL_US_VALUE "1000")
endif()

set(AEB_MAX_SAMPLE_BYTES_VALUE $ENV{AEB_MAX_SAMPLE_BYTES})
if(DEFINED AEB_MAX_SAMPLE_BYTES)
    set(AEB_MAX_SAMPLE_BYTES_VALUE ${AEB_MAX_SAMPLE_BYTES})
endif()
if(NOT AEB_MAX_SAMPLE_BYTES_VALUE)
    set(AEB_MAX_SAMPLE_BYTES_VALUE "8192")
endif()

set(AEB_MAX_BURSTS_VALUE $ENV{AEB_MAX_BURSTS})
if(DEFINED AEB_MAX_BURSTS)
    set(AEB_MAX_BURSTS_VALUE ${AEB_MAX_BURSTS})
endif()
if(NOT AEB_MAX_BURSTS_VALUE)
    set(AEB_MAX_BURSTS_VALUE "256")
endif()

set(AEB_RAW_CHUNK_BYTES_VALUE $ENV{AEB_RAW_CHUNK_BYTES})
if(DEFINED AEB_RAW_CHUNK_BYTES)
    set(AEB_RAW_CHUNK_BYTES_VALUE ${AEB_RAW_CHUNK_BYTES})
endif()
if(NOT AEB_RAW_CHUNK_BYTES_VALUE)
    set(AEB_RAW_CHUNK_BYTES_VALUE "512")
endif()

set(AEB_TRIAL_GAP_MS_VALUE $ENV{AEB_TRIAL_GAP_MS})
if(DEFINED AEB_TRIAL_GAP_MS)
    set(AEB_TRIAL_GAP_MS_VALUE ${AEB_TRIAL_GAP_MS})
endif()
if(NOT AEB_TRIAL_GAP_MS_VALUE)
    set(AEB_TRIAL_GAP_MS_VALUE "250")
endif()

set(AEB_DUMP_RAW_HEX_VALUE $ENV{AEB_DUMP_RAW_HEX})
if(DEFINED AEB_DUMP_RAW_HEX)
    set(AEB_DUMP_RAW_HEX_VALUE ${AEB_DUMP_RAW_HEX})
endif()

    set(AEB_DUMP_RAW_HEX_VALUE "0")
endif()

set(AEB_DUMP_SEED_VALUE $ENV{AEB_DUMP_SEED})
if(DEFINED AEB_DUMP_SEED)
    set(AEB_DUMP_SEED_VALUE ${AEB_DUMP_SEED})
endif()
if(NOT AEB_DUMP_SEED_VALUE)
    set(AEB_DUMP_SEED_VALUE "0")
endif()

zephyr_include_directories(include)
target_include_directories(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

target_sources(app PRIVATE
    src/main.c
    src/wifi.c
)

target_compile_definitions(app PRIVATE
    WIFI_SSID="${WIFI_SSID_VALUE}"
    WIFI_PASS="${WIFI_PASS_VALUE}"
    AEB_UDP_PORT=${AEB_UDP_PORT_VALUE}
    AEB_GATEWAY_IP="${AEB_GATEWAY_IP_VALUE}"
    AEB_GATEWAY_PORT=${AEB_GATEWAY_PORT_VALUE}
    AEB_CLIENT_INITIATED=${AEB_CLIENT_INITIATED_VALUE}
    AEB_CLIENT_TRIALS=${AEB_CLIENT_TRIALS_VALUE}
    AEB_BURST_COUNT=${AEB_BURST_COUNT_VALUE}
    AEB_INTERVAL_US=${AEB_INTERVAL_US_VALUE}
    AEB_MAX_SAMPLE_BYTES=${AEB_MAX_SAMPLE_BYTES_VALUE}
    AEB_MAX_BURSTS=${AEB_MAX_BURSTS_VALUE}
    AEB_RAW_CHUNK_BYTES=${AEB_RAW_CHUNK_BYTES_VALUE}
    AEB_TRIAL_GAP_MS=${AEB_TRIAL_GAP_MS_VALUE}
    AEB_DUMP_RAW_HEX=${AEB_DUMP_RAW_HEX_VALUE}
    AEB_DUMP_SEED=${AEB_DUMP_SEED_VALUE}
)
